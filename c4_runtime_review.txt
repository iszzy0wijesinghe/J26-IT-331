from datetime import datetime

from pydantic import Field, model_validator

from shared.contracts.common import TimeAwareContractModel

class MeasurementEntry(TimeAwareContractModel):
    """One measurement contained inside a telemetry packet.

    Sensor type remains a controlled string for now because the final
    sensor registry is explicitly deferred by Integration Contract v1.
    """

    sensor_instance_id: str = Field(min_length=1)
    sensor_type: str = Field(min_length=1)
    channel: str | None = None

    raw_value: float | int | None = None
    value: float | None = None
    unit: str | None = None

    calibration_version: str | None = None

    @model_validator(mode="after")
    def require_measurement_value(self) -> "MeasurementEntry":
        if self.raw_value is None and self.value is None:
            raise ValueError(
                "A measurement must contain raw_value, value, or both"
            )

        return self


class DeviceHealth(TimeAwareContractModel):
    """Device/system-health metadata carried with telemetry.

    This is intentionally separate from environmental measurements.
    """

    battery_voltage: float | None = None
    supply_current: float | None = None
    restart_count: int | None = Field(default=None, ge=0)
    sensor_read_error: bool | None = None
    sd_status: str | None = None

    extra_metrics: dict[str, Any] = Field(default_factory=dict)


class TransportMetadata(TimeAwareContractModel):
    """Transport/network metadata associated with a packet."""

    rssi_dbm: float | None = None
    snr_db: float | None = None
    retry_count: int | None = Field(default=None, ge=0)
    gateway_id: str | None = None

    extra_metadata: dict[str, Any] = Field(default_factory=dict)


class SecurityMetadata(TimeAwareContractModel):
    """Non-secret security metadata associated with telemetry.

    Secret keys, passwords, tokens and raw credentials must never be
    placed in this contract.
    """

    credential_id: str | None = None
    integrity_present: bool | None = None
    freshness_counter: int | None = Field(default=None, ge=0)
    session_id: str | None = None
    firmware_hash: str | None = None
    config_hash: str | None = None
    # Opaque protocol evidence; C4 must verify these through configured verifiers.
    attestation_evidence: dict[str, Any] | None = None
    provenance_evidence: dict[str, Any] | None = None

    extra_metadata: dict[str, Any] = Field(default_factory=dict)


class TelemetryPacket(TimeAwareContractModel):
    """Normalized gateway-to-ingestion telemetry packet."""

    schema_version: str = Field(min_length=1)

    packet_id: str = Field(min_length=1)
    device_id: str = Field(min_length=1)
    zone_id: str = Field(min_length=1)

    boot_id: str | None = None
    sequence_no: int | None = Field(default=None, ge=0)

    observed_at: datetime
    received_at: datetime

    message_type: str = Field(default="TELEMETRY", min_length=1)

    measurements: list[MeasurementEntry] = Field(default_factory=list)

    device_health: DeviceHealth = Field(default_factory=DeviceHealth)
    transport: TransportMetadata = Field(default_factory=TransportMetadata)
    security: SecurityMetadata = Field(default_factory=SecurityMetadata)

    source_origin: SourceOrigin
    test_run_id: str | None = None

    correlation_id: str | None = None

"""Deterministic baseline telemetry generation for GeoNex."""

import random
from dataclasses import dataclass
from datetime import UTC, datetime, timedelta

from shared.contracts.telemetry import TelemetryPacket
from shared.enums.domain import OperatingMode, SourceOrigin


@dataclass(frozen=True)
class SimulationConfig:
    """Configuration for a reproducible simulation run."""

    seed: int = 42
    zone_id: str = "ZONE_01"
    device_id: str = "PIPE_A1"
    start_time: datetime = datetime(2026, 10, 10, 0, 0, tzinfo=UTC)
    packet_count: int = 10
    operating_mode: OperatingMode = OperatingMode.IDLE


MODE_INTERVALS_SECONDS = {
    OperatingMode.DEEP_SLEEP: 900,
    OperatingMode.IDLE: 300,
    OperatingMode.RISK: 60,
    OperatingMode.WARNING: 15,
    OperatingMode.EMERGENCY: 5,
}


def generate_baseline_packets(
    config: SimulationConfig,
) -> list[TelemetryPacket]:
    """Generate repeatable baseline telemetry packets.

    This is a synthetic engineering fixture, not a calibrated
    physical landslide simulation or research dataset.
    """

    if config.start_time.utcoffset() is None:
        raise ValueError("start_time must be timezone-aware")

    if config.packet_count < 0:
        raise ValueError("packet_count must not be negative")

    if not config.zone_id or not config.device_id:
        raise ValueError("zone_id and device_id must not be empty")

    rng = random.Random(config.seed)

    interval_seconds = MODE_INTERVALS_SECONDS[config.operating_mode]

    packets: list[TelemetryPacket] = []

    for index in range(config.packet_count):
        observed_at = config.start_time + timedelta(
            seconds=index * interval_seconds
        )

        received_at = observed_at + timedelta(seconds=2)

        soil_moisture = round(35.0 + rng.uniform(-2.0, 2.0), 3)
        temperature = round(27.0 + rng.uniform(-1.5, 1.5), 3)
        pressure = round(1013.0 + rng.uniform(-3.0, 3.0), 3)

        packet = TelemetryPacket(
            schema_version="1.0",
            packet_id=(
                f"SIM_{config.seed}_{config.zone_id}_"
                f"{config.device_id}_{index:06d}"
            ),
            device_id=config.device_id,
            zone_id=config.zone_id,
            boot_id=f"SIM_BOOT_{config.seed}",
            sequence_no=index,
            observed_at=observed_at,
            received_at=received_at,
            message_type="TELEMETRY",
            measurements=[
                {
                    "sensor_instance_id": f"{config.device_id}_SOIL_01",
                    "sensor_type": "SOIL_MOISTURE",
                    "value": soil_moisture,
                    "unit": "%",
                },
                {
                    "sensor_instance_id": f"{config.device_id}_TEMP_01",
                    "sensor_type": "TEMPERATURE",
                    "value": temperature,
                    "unit": "C",
                },
                {
                    "sensor_instance_id": f"{config.device_id}_PRESSURE_01",
                    "sensor_type": "ATMOSPHERIC_PRESSURE",
                    "value": pressure,
                    "unit": "hPa",
                },
            ],
            source_origin=SourceOrigin.SIMULATED,
            test_run_id=f"SIM_RUN_{config.seed}",
            correlation_id=f"SIM_CORR_{config.seed}_{index:06d}",
        )

        packets.append(packet)

    return packets

        "device_id",

        "received_timestamp",

        "firmware_hash",
        "config_hash",

        "label_untrusted_binary",
        "c4_decision",
        "c4_trust_score",
        "reason_code",
        "split",

    # Behavioral feature fields to INCLUDE
    BEHAVIORAL_FIELDS = {
        # B0 - Timing

        "freshness_status",
        # B1 - Radio
        "rssi_dbm",
        "rssi_deviation_db",

        "battery_pct",
        "battery_voltage_v",
        "current_ma",
        "reboot_count_24h",
        "sensor_fault_flag",
        # B4 - Path (derived from comparison)
        "zone_registered",
        "zone_reported",
        "hub_registered",
        "hub_id",
    }

    TRAINING_FEATURE_FIELDS = (
        "tx_interval_s",
        "rx_delay_s",
        "freshness_status_encoded",
        "rssi_dbm",
        "rssi_deviation_db",
        "snr_db",
        "retry_count",
        "packet_size_bytes",
        "battery_pct",
        "battery_voltage_v",
        "current_ma",
        "reboot_count_24h",
        "sensor_fault_flag",
        "rolling_interval_mean",
        "rolling_interval_std",
        "rolling_rssi_mean",
        "rolling_rssi_std",
        "rolling_retry_mean",
    )

    def __init__(self) -> None:
        self._freshness_status_encoding = {
            "fresh": 0,
            "delayed": 1,
            "stale": 2,
        }
        # Expected intervals per mode (in seconds) - reserved for future use
        # These should be configured based on actual sampling policies
        self._mode_expected_intervals = {
            "IDLE": 30.0,  # Normal low-power mode
            "ACTIVE": 10.0,  # Normal active sampling
            "WARNING": 5.0,  # High-risk mode, frequent sampling
            "CRITICAL": 1.0,  # Emergency mode, very frequent sampling
            "DEEP_SLEEP": 300.0,  # Very low power mode
            "UNKNOWN": 10.0,  # Default fallback
        }

    def extract_from_dataframe(self, df: pd.DataFrame) -> pd.DataFrame:
        """Extract behavioral features from dataframe.

        Args:
            df: Input dataframe with C4 dataset columns

        Returns:
            DataFrame with only behavioral features
        """
        # Create a copy to avoid modifying original
        features_df = df.copy()

        # Encode freshness_status
        features_df["freshness_status_encoded"] = features_df[
            "freshness_status"
        ].map(self._freshness_status_encoding).fillna(0)

        # Select only behavioral fields
        selected_columns = [
            # B0 - Timing
            "tx_interval_s",
            "rx_delay_s",
            "freshness_status_encoded",
            # B1 - Radio
            "rssi_dbm",
            "rssi_deviation_db",
            "snr_db",
            "retry_count",
            # B2 - Packet
            "packet_size_bytes",
            # B3 - Health
            "battery_pct",
            "battery_voltage_v",
            "current_ma",
            "reboot_count_24h",
            "sensor_fault_flag",
        ]

        features_df = features_df[selected_columns]

        return features_df

    def add_mode_aware_features(
        self,
        df: pd.DataFrame,
        mode_col: str = "mode",
    ) -> pd.DataFrame:
        """Add mode-aware interval deviation features.

        Reserved for future use when mode information is available.

        Args:
            df: Input dataframe with mode and tx_interval_s columns
            mode_col: Name of mode column

        Returns:
            DataFrame with added mode-aware features (empty for now)
        """
        df = df.copy()

        # Mode-aware features are not currently supported in the dataset
        # This method is reserved for future implementation when mode information
        # is available in the telemetry protocol
        return df

    def add_historical_features(
        self,
        df: pd.DataFrame,
        device_id_col: str = "device_id",
        window: int = 10,
    ) -> pd.DataFrame:
        """Add rolling historical features per device.

        Args:
            df: Input dataframe with device_id and behavioral features
            device_id_col: Name of device_id column
            window: Rolling window size

        Returns:
            DataFrame with added historical features
        """
        if device_id_col not in df.columns:
            raise ValueError(f"{device_id_col} is required to calculate historical features")
        if "device_timestamp" not in df.columns:
            raise ValueError("device_timestamp is required to calculate historical features")

        timestamps = pd.to_datetime(
            df["device_timestamp"], format="mixed", utc=True, errors="coerce"
        )
        if timestamps.isna().any():
            raise ValueError("device_timestamp contains missing or invalid values")

        original_index = df.index
        timestamp_key = "__c4_history_timestamp"
        position_key = "__c4_history_position"
        while timestamp_key in df.columns:
            timestamp_key = f"_{timestamp_key}"
        while position_key in df.columns or position_key == timestamp_key:
            position_key = f"_{position_key}"

        df = df.copy().reset_index(drop=True)
        df[position_key] = range(len(df))
        df[timestamp_key] = timestamps.array
        df = df.sort_values([device_id_col, timestamp_key], kind="stable")

        # Treat tied timestamps as one step so peers cannot affect one another.
        feature_sources = {
            "tx_interval_s": ("rolling_interval_mean", "rolling_interval_std"),
            "rssi_dbm": ("rolling_rssi_mean", "rolling_rssi_std"),
            "retry_count": ("rolling_retry_mean", None),
        }
        for _, device_group in df.groupby(device_id_col, sort=False):
            timestamp_groups = device_group.groupby(timestamp_key, sort=True)
            for source_column, (mean_column, std_column) in feature_sources.items():
                if source_column not in df.columns:
                    continue

                timestamp_means = timestamp_groups[source_column].mean()
                previous_values = timestamp_means.shift(1)
                rolling_mean = previous_values.rolling(
                    window=window, min_periods=1
                ).mean()
                df.loc[device_group.index, mean_column] = device_group[timestamp_key].map(
                    rolling_mean
                ).to_numpy()

                if std_column is not None:
                    rolling_std = previous_values.rolling(
                        window=window, min_periods=1
                    ).std()
                    df.loc[device_group.index, std_column] = device_group[timestamp_key].map(
                        rolling_std
                    ).to_numpy()

        # Fill NaN values with 0 for first rows
        historical_cols = [
            "rolling_interval_mean",
            "rolling_interval_std",
            "rolling_rssi_mean",
            "rolling_rssi_std",
            "rolling_retry_mean",
        ]
        for col in historical_cols:
            if col in df.columns:
                df[col] = df[col].fillna(0)

        positions = df.pop(position_key).to_numpy()
        df = df.drop(columns=timestamp_key)
        df.index = original_index.take(positions)
        return df

    def extract_for_training(
        self,
        df: pd.DataFrame,
        add_historical: bool = True,
        historical_window: int = 10,
    ) -> tuple[pd.DataFrame, pd.Series]:
        """Extract features and labels for ML training.

        Args:
            df: Input C4 dataset
            add_historical: Whether to add rolling historical features
            historical_window: Window size for historical features

        Returns:
            Tuple of (features DataFrame, labels Series)
        """
        # Accept either raw C4 rows or an already-extracted behavioral dataset.
        if self.BEHAVIORAL_FIELDS.issubset(df.columns):
            features_df = self.extract_from_dataframe(df)
        else:
            available_fields = [
                column for column in self.TRAINING_FEATURE_FIELDS if column in df.columns
            ]
            if not available_fields:
                raise ValueError("Input dataframe has no recognized C4 behavioral features")
            features_df = df[available_fields].copy()

        # Add historical features if requested
        history_sources = {"tx_interval_s", "rssi_dbm", "retry_count"} & set(df.columns)
        if add_historical and history_sources:
            if "device_id" not in df.columns:
                raise ValueError("device_id is required to calculate historical features")
            history_input_columns = [
                column
                for column in (
                    "device_id",
                    "device_timestamp",
                    "tx_interval_s",
                    "rssi_dbm",
                    "retry_count",
                )
                if column in df.columns
            ]
            history_features = self.add_historical_features(
                df[history_input_columns],
                device_id_col="device_id",
                window=historical_window,
            )
            historical_columns = [
                "rolling_interval_mean",
                "rolling_interval_std",
                "rolling_rssi_mean",
                "rolling_rssi_std",
                "rolling_retry_mean",
            ]
            features_df = features_df.drop(
                columns=[column for column in historical_columns if column in features_df]
            )
            features_df = features_df.join(
                history_features[
                    [column for column in historical_columns if column in history_features]
                ]
            )

        feature_columns = [
            column for column in self.TRAINING_FEATURE_FIELDS if column in features_df
        ]
        features_df = features_df[feature_columns]

        # Extract labels
        labels = df["trust_label"].copy()

        return features_df, labels

    def create_behavioral_dataset(
        self,
        input_path: str,
        output_path: str,
        add_historical: bool = True,
        historical_window: int = 10,
    ) -> None:
        """Create clean behavioral dataset from C4_Trust.csv.

        Args:
            input_path: Path to C4_Trust.csv
            output_path: Path to save behavioral dataset
            add_historical: Whether to add historical features
            historical_window: Window size for historical features
        """
        # Load dataset
        df = pd.read_csv(input_path)

        # Extract features and labels
        features_df, labels = self.extract_for_training(
            df,
            add_historical=add_historical,
            historical_window=historical_window,
        )

        # Combine features with labels and metadata
        output_df = features_df.copy()
        output_df["trust_label"] = labels
        output_df["scenario"] = df["scenario"]
        output_df["device_id"] = df["device_id"]
        output_df["device_timestamp"] = df["device_timestamp"]

        # Save to CSV
        output_df.to_csv(output_path, index=False)

        print(f"Behavioral dataset saved to {output_path}")
        print(f"Shape: {output_df.shape}")
        print(f"Features: {len(features_df.columns)}")
        print(f"Label distribution:\n{labels.value_counts()}")

class ModelSchemaError(ValueError):
    """A model artifact or feature frame does not match the expected schema."""


def _validate_feature_frame(
    features: pd.DataFrame,
    expected_feature_names: Sequence[str],
) -> None:
    expected = list(expected_feature_names)
    actual = list(features.columns)
    if actual != expected:
        raise ModelSchemaError(
            f"Feature schema mismatch: expected ordered columns {expected}, got {actual}"
        )
    if not all(pd.api.types.is_numeric_dtype(dtype) for dtype in features.dtypes):
        raise ModelSchemaError("All model features must be numeric")
    try:
        values = features.to_numpy(dtype=float)
    except (TypeError, ValueError) as exc:
        raise ModelSchemaError("Model features must be numeric") from exc
    if not np.isfinite(values).all():
        raise ModelSchemaError("Model features must not contain missing or non-finite values")

            path: Path to save model
        """
        path = Path(path)
        path.parent.mkdir(parents=True, exist_ok=True)
        created = False
        try:
            with path.open("xb") as artifact_file:
                created = True
                joblib.dump(self, artifact_file)
        except Exception:
            if created:
                path.unlink(missing_ok=True)
            raise
        print(f"Model saved to {path}")

    def validate_feature_schema(self, expected_feature_names: Sequence[str]) -> None:
        if self.feature_names != list(expected_feature_names):
            raise ModelSchemaError(
                "Model artifact feature schema does not match the requested C4 schema"
            )

    def predict(self, features: pd.DataFrame) -> np.ndarray:
        _validate_feature_frame(features, self.feature_names)
        if self.model_type == ModelType.ISOLATION_FOREST:
            if not {"Trusted", "Suspicious"}.issubset(self.label_encoder.classes_):
                raise ModelSchemaError(
                    "Isolation Forest can only map anomalies to SUSPICIOUS and "
                    "normal points to TRUSTED"
                )
            values = self.scaler.transform(features) if self.scaler is not None else features.values
            return np.where(self.model.predict(values) == 1, "Trusted", "Suspicious")

        values = self.scaler.transform(features) if self.scaler is not None else features.values
        encoded = self.model.predict(values)
        return self.label_encoder.inverse_transform(encoded)

    def predict_proba(self, features: pd.DataFrame) -> np.ndarray:
        _validate_feature_frame(features, self.feature_names)
        if self.model_type == ModelType.ISOLATION_FOREST:
            raise ModelSchemaError(
                "Isolation Forest has no calibrated class probabilities for C4 fusion"
            )
        values = self.scaler.transform(features) if self.scaler is not None else features.values
        probabilities = self.model.predict_proba(values)
        if probabilities.shape[1] != len(self.label_encoder.classes_):
            raise ModelSchemaError("Model probability columns do not match encoded labels")
        return probabilities

    def behavioural_anomaly_score(self, features: pd.DataFrame) -> np.ndarray:
        if self.model_type == ModelType.ISOLATION_FOREST:
            raise ModelSchemaError(
                "Isolation Forest requires a separately calibrated anomaly threshold"
            )
        try:
            trusted_index = list(self.label_encoder.classes_).index("Trusted")
        except ValueError as exc:
            raise ModelSchemaError("Model artifact has no Trusted class") from exc
        return 1.0 - self.predict_proba(features)[:, trusted_index]

    @classmethod
    def load(
        cls,
        path: Path,
        *,
        expected_feature_names: Sequence[str] | None = None,
    ) -> TrainedModel:
        """Load model from disk.

        Args:
            path: Path to load model from

        Returns:
            TrainedModel instance
        """
        trained = joblib.load(path)
        if not isinstance(trained, cls):
            raise ModelSchemaError("Artifact is not a C4 TrainedModel")
        if expected_feature_names is not None:
            trained.validate_feature_schema(expected_feature_names)
        return trained


class C4BehavioralModelRunner:
    """Load a compatible artifact and score one packet through a feature adapter.

    This interface provides safe model loading with explicit schema validation.
    If the model cannot be loaded or has an incompatible schema, the load method
    will raise ModelSchemaError with a clear error message.
    """

    def __init__(
        self,
        model: TrainedModel,
        feature_builder: Callable[[TelemetryPacket], pd.DataFrame],
    ) -> None:
        if model.model_type == ModelType.ISOLATION_FOREST:
            raise ModelSchemaError(
                "Isolation Forest is not calibrated for C4 trust-score fusion"
            )
        self._model = model
        self._feature_builder = feature_builder

    @classmethod
    def load(
        cls,
        path: Path,
        expected_feature_names: Sequence[str],
        feature_builder: Callable[[TelemetryPacket], pd.DataFrame],
    ) -> C4BehavioralModelRunner:
        """Load a model from disk with schema validation.

        Args:
            path: Path to model artifact
            expected_feature_names: Expected feature schema
            feature_builder: Function to extract features from a TelemetryPacket

        Returns:
            C4BehavioralModelRunner instance

        Raises:
            FileNotFoundError: If model file does not exist
            ModelSchemaError: If model artifact is corrupted or has incompatible schema
        """
        try:
            model = TrainedModel.load(path, expected_feature_names=expected_feature_names)
        except FileNotFoundError as exc:
            raise FileNotFoundError(
                f"Model artifact not found at {path}. "
                "Cannot perform behavioral scoring without a trained model."
            ) from exc
        except (OSError, ValueError, TypeError, KeyError) as exc:
            raise ModelSchemaError(
                f"Failed to load model artifact from {path}: {exc}. "
                "The model may be corrupted or incompatible with the current schema."
            ) from exc
        return cls(model, feature_builder)

    def __call__(self, packet: TelemetryPacket) -> float:
        """Score a single telemetry packet.

        Args:
            packet: TelemetryPacket to score

        Returns:
            Behavioral anomaly score (0.0 = normal, 1.0 = anomalous)

        Raises:
            ModelSchemaError: If feature extraction fails or schema mismatch
        """
        try:
            features = self._feature_builder(packet)
        except Exception as exc:
            raise ModelSchemaError(
                f"Feature extraction failed for packet {packet.packet_id}: {exc}"
            ) from exc

        if len(features) != 1:
            raise ModelSchemaError("A packet feature builder must return exactly one row")

        try:
            return float(self._model.behavioural_anomaly_score(features)[0])
        except Exception as exc:
            raise ModelSchemaError(
                f"Behavioral scoring failed for packet {packet.packet_id}: {exc}"
            ) from exc


class BaselineModel:
    """Baseline models for C4 behavioral classification."""

    def __init__(
        self,
        model_type: ModelType = ModelType.LOGISTIC_REGRESSION,
        random_state: int = 42,
    ) -> None:
        """Initialize baseline model.

        Args:
            model_type: Type of model to use
            random_state: Random seed for reproducibility
        """
        self.model_type = model_type
        self.random_state = random_state
        self.model: Any = None
        self.scaler: StandardScaler | None = None
        self.label_encoder: LabelEncoder = LabelEncoder()
        self.feature_names: list[str] = []

        self._initialize_model()

    def _initialize_model(self) -> None:
        """Initialize the underlying model."""
        if self.model_type == ModelType.LOGISTIC_REGRESSION:
            self.model = LogisticRegression(
                random_state=self.random_state,
                max_iter=1000,
            )
            self.scaler = StandardScaler()
        elif self.model_type == ModelType.RANDOM_FOREST:
            self.model = RandomForestClassifier(
                n_estimators=100,
                random_state=self.random_state,
                n_jobs=-1,
            )
            self.scaler = None
        elif self.model_type == ModelType.ISOLATION_FOREST:
            self.model = IsolationForest(
                n_estimators=100,
                random_state=self.random_state,
                contamination=0.1,
            )
            self.scaler = None
        elif self.model_type == ModelType.XGBOOST:
            if not XGBOOST_AVAILABLE:
                raise ImportError(
                    "XGBoost is not installed. Install it with: pip install xgboost"
                )
            self.model = xgb.XGBClassifier(
                n_estimators=100,
                random_state=self.random_state,
                n_jobs=-1,
                use_label_encoder=False,
                eval_metric="mlogloss",
            )
            self.scaler = None
        else:
            raise ValueError(f"Unsupported model type: {self.model_type}")

    def fit(
        self,
        X_train: pd.DataFrame,
        y_train: pd.Series,
        X_val: pd.DataFrame | None = None,
        y_val: pd.Series | None = None,
    ) -> TrainedModel:
        """Train the model.

        Args:
            X_train: Training features
            y_train: Training labels
            X_val: Validation features (optional)
            y_val: Validation labels (optional)

        Returns:
            TrainedModel instance
        """
        self.feature_names = list(X_train.columns)
        _validate_feature_frame(X_train, self.feature_names)

        # Encode labels
        y_train_encoded = self.label_encoder.fit_transform(y_train)

        # Scale features if needed
        if self.scaler is not None:
            X_train_scaled = self.scaler.fit_transform(X_train)
        else:
            X_train_scaled = X_train.values

        # Train model
        if self.model_type == ModelType.ISOLATION_FOREST:
            # Isolation Forest is unsupervised, fit on features only
            self.model.fit(X_train_scaled)
            # Predictions: -1 (anomaly) or 1 (normal)
            # Map to labels: -1 -> Untrusted/Suspicious, 1 -> Trusted
        else:
            self.model.fit(X_train_scaled, y_train_encoded)

        # Evaluate on validation set if provided
        metrics = None
        if X_val is not None and y_val is not None:
            metrics = self.evaluate(X_val, y_val)

        # Create trained model container
        trained = TrainedModel(
            model=self.model,
            scaler=self.scaler,
            label_encoder=self.label_encoder,
            model_type=self.model_type,
            feature_names=self.feature_names,
            metrics=metrics,
            training_metadata={
                "random_state": self.random_state,
                "n_train_samples": len(X_train),
                "n_features": len(self.feature_names),
            },
        )

        return trained

    def predict(self, X: pd.DataFrame) -> np.ndarray:
        """Make predictions.

        Args:
            X: Features

        Returns:
            Predicted labels
        """
        _validate_feature_frame(X, self.feature_names)
        if self.scaler is not None:
            X_scaled = self.scaler.transform(X)
        else:
            X_scaled = X.values

        if self.model_type == ModelType.ISOLATION_FOREST:
            predictions = self.model.predict(X_scaled)
            if not {"Trusted", "Suspicious"}.issubset(self.label_encoder.classes_):
                raise ModelSchemaError(
                    "Isolation Forest can only map anomalies to SUSPICIOUS and "
                    "normal points to TRUSTED"
                )
            return np.where(predictions == 1, "Trusted", "Suspicious")
        else:
            encoded_pred = self.model.predict(X_scaled)

        # Decode labels
        labels = self.label_encoder.inverse_transform(encoded_pred)
        return labels

    def predict_proba(self, X: pd.DataFrame) -> np.ndarray:
        """Get prediction probabilities.

        Args:
            X: Features

        Returns:
            Prediction probabilities
        """
        _validate_feature_frame(X, self.feature_names)
        if self.model_type == ModelType.ISOLATION_FOREST:
            raise ModelSchemaError(
                "Isolation Forest has no calibrated class probabilities for C4 fusion"
            )
        if self.scaler is not None:
            X_scaled = self.scaler.transform(X)
        else:
            X_scaled = X.values
        return self.model.predict_proba(X_scaled)

    def evaluate(self, X: pd.DataFrame, y: pd.Series) -> ModelMetrics:
        """Evaluate model on dataset.

        Args:
            X: Features
            y: True labels

        Returns:
            ModelMetrics
        """
        y_pred = self.predict(X)

        # Compute metrics
        accuracy = (y_pred == y).mean()

        precision, recall, f1, support = precision_recall_fscore_support(
            y, y_pred, average=None, zero_division=0
        )

        precision_macro, recall_macro, f1_macro, _ = precision_recall_fscore_support(
            y, y_pred, average="macro", zero_division=0
        )

        # Per-class metrics
        classes = self.label_encoder.classes_
        precision_per_class = {cls: prec for cls, prec in zip(classes, precision, strict=False)}
        recall_per_class = {cls: rec for cls, rec in zip(classes, recall, strict=False)}
        f1_per_class = {cls: f for cls, f in zip(classes, f1, strict=False)}

        # Confusion matrix
        cm = confusion_matrix(y, y_pred, labels=classes)

        return ModelMetrics(
            accuracy=accuracy,
            precision_macro=precision_macro,
            recall_macro=recall_macro,
            f1_macro=f1_macro,
            precision_per_class=precision_per_class,
            recall_per_class=recall_per_class,
            f1_per_class=f1_per_class,
            confusion_matrix=cm,
        )

    def get_feature_importance(self) -> dict[str, float] | None:
        """Get feature importance if available.

        Returns:
            Dictionary of feature names to importance scores, or None
        """
        if self.model_type == ModelType.RANDOM_FOREST:
            importances = self.model.feature_importances_
            return dict(zip(self.feature_names, importances, strict=True))
        elif self.model_type == ModelType.LOGISTIC_REGRESSION:
            # For multi-class, get mean absolute coefficient across classes
            coef = np.abs(self.model.coef_)
            if coef.ndim > 1:
                importances = coef.mean(axis=0)
            else:
                importances = coef
            return dict(zip(self.feature_names, importances, strict=True))
        else:
            return None


def train_baseline_model(
    X_train: pd.DataFrame,
    y_train: pd.Series,
    X_val: pd.DataFrame,
    y_val: pd.Series,
    model_type: ModelType = ModelType.LOGISTIC_REGRESSION,
    output_path: Path | None = None,
) -> TrainedModel:
    """Train a baseline model.

    Args:
        X_train: Training features
        y_train: Training labels
        X_val: Validation features
        y_val: Validation labels
        model_type: Type of model to train
        output_path: Path to save model (optional)

    Returns:
        TrainedModel instance
    """
    print(f"Training {model_type.value} model...")

    model = BaselineModel(model_type=model_type)
    trained = model.fit(X_train, y_train, X_val, y_val)

    if trained.metrics:
        print("\nValidation Metrics:")
        print(f"  Accuracy: {trained.metrics.accuracy:.4f}")
        print(f"  Macro F1: {trained.metrics.f1_macro:.4f}")
        print("\nPer-class F1:")
        for cls, f1 in trained.metrics.f1_per_class.items():
            print(f"  {cls}: {f1:.4f}")

    if output_path:
        trained.save(output_path)

    return trained

from __future__ import annotations

import hashlib
import hmac
import json

from components.c4_trust.registry import DeviceCredential, DeviceRecord, DeviceRegistry
from shared.contracts import C4TrustAssessment, TelemetryPacket

EvidenceVerifier = Callable[[TelemetryPacket, DeviceRecord | None], bool | None]
BehaviouralScorer = Callable[[TelemetryPacket], float | None]


@dataclass
class RecoveryContext:
    """Context for authorized C2 recovery replay.

    This distinguishes legitimate C2-requested historical data replay
    from malicious replay attacks.
    """

    recovery_request_id: str
    authorized_device_id: str
    authorized_start_time: datetime
    authorized_end_time: datetime
    created_at: datetime = field(default_factory=lambda: datetime.now(UTC))
    is_active: bool = True


class C4TrustService:
    def __init__(
        self,
        credentials_or_registry: Mapping[str, DeviceCredential] | DeviceRegistry,
        *,
        max_age_seconds: float = 30.0,
        future_tolerance_seconds: float = 5.0,
        suspicious_score_threshold: float = 0.7,
        attestation_verifier: EvidenceVerifier | None = None,
        provenance_verifier: EvidenceVerifier | None = None,
        behavioural_scorer: BehaviouralScorer | None = None,
        allow_deterministic_only: bool = False,
        require_sequence: bool = True,
        require_firmware_approval: bool = False,
        require_config_approval: bool = False,
    ) -> None:
        if max_age_seconds <= 0:
            raise ValueError("max_age_seconds must be positive")
        if future_tolerance_seconds < 0:
            raise ValueError("future_tolerance_seconds cannot be negative")
        if not 0.0 <= suspicious_score_threshold <= 1.0:
            raise ValueError("suspicious_score_threshold must be between 0 and 1")

        # Support both legacy credentials mapping and new DeviceRegistry
        if isinstance(credentials_or_registry, DeviceRegistry):
            self._registry = credentials_or_registry
            self._credentials = {}
        else:
            self._registry = None
            self._credentials = dict(credentials_or_registry)

        self._max_age = timedelta(seconds=max_age_seconds)
        self._future_tolerance = timedelta(seconds=future_tolerance_seconds)
        self._suspicious_score_threshold = suspicious_score_threshold
        self._attestation_verifier = attestation_verifier
        self._provenance_verifier = provenance_verifier
        self._behavioural_scorer = behavioural_scorer
        self._allow_deterministic_only = allow_deterministic_only
        self._require_sequence = require_sequence
        self._require_firmware_approval = require_firmware_approval
        self._require_config_approval = require_config_approval
        self._seen_packets: set[tuple[str, str]] = set()
        self._last_sequence: dict[tuple[str, str | None], int] = {}
        self._recovery_contexts: dict[str, RecoveryContext] = {}
        self._lock = threading.Lock()

    @staticmethod
    def _canonical_packet(packet: TelemetryPacket) -> bytes:
        payload = packet.model_dump(mode="json")
        return json.dumps(
            payload,
            sort_keys=True,
            separators=(",", ":"),
            ensure_ascii=False,
        ).encode("utf-8")

    @classmethod
    def sign_packet(
        cls,
        packet: TelemetryPacket,
        secret: bytes,
    ) -> str:
        return hmac.new(
            secret,
            cls._canonical_packet(packet),
            hashlib.sha256,
        ).hexdigest()

    def create_recovery_context(
        self,
        recovery_request_id: str,
        authorized_device_id: str,
        authorized_start_time: datetime,
        authorized_end_time: datetime,
    ) -> RecoveryContext:
        """Create an authorized recovery context for C2 replay.

        Args:
            recovery_request_id: Unique identifier for the recovery request
            authorized_device_id: Device authorized to replay data
            authorized_start_time: Start of authorized time window
            authorized_end_time: End of authorized time window

        Returns:
            RecoveryContext
        """
        for name, value in (
            ("authorized_start_time", authorized_start_time),
            ("authorized_end_time", authorized_end_time),
        ):
            if value.tzinfo is None or value.utcoffset() is None:
                raise ValueError(f"{name} must be timezone-aware")
        if authorized_end_time < authorized_start_time:
            raise ValueError("authorized_end_time must not precede authorized_start_time")

        context = RecoveryContext(
            recovery_request_id=recovery_request_id,
            authorized_device_id=authorized_device_id,
            authorized_start_time=authorized_start_time,
            authorized_end_time=authorized_end_time,
        )

        with self._lock:
            self._recovery_contexts[recovery_request_id] = context

        return context

    def revoke_recovery_context(self, recovery_request_id: str) -> bool:
        """Revoke an authorized recovery context.

        Args:
            recovery_request_id: Recovery request ID to revoke

        Returns:
            True if context was found and revoked, False otherwise
        """
        with self._lock:
            if recovery_request_id in self._recovery_contexts:
                context = self._recovery_contexts[recovery_request_id]
                context.is_active = False
                return True
            return False

    def _check_authorized_replay(
        self,
        packet: TelemetryPacket,
        recovery_request_id: str | None,
    ) -> bool:
        """Check if replay is authorized under a recovery context.

        Args:
            packet: Telemetry packet to check
            recovery_request_id: Recovery request ID from packet metadata

        Returns:
            True if replay is authorized, False otherwise
        """
        if not recovery_request_id:
            return False

        with self._lock:
            context = self._recovery_contexts.get(recovery_request_id)

        if context is None or not context.is_active:
            return False

        # Check device authorization
        if packet.device_id != context.authorized_device_id:
            return False

        # Check time window
        if not (
            context.authorized_start_time <= packet.observed_at <= context.authorized_end_time
        ):
            return False

        return True

    @staticmethod
    def _verify_evidence(
        verifier: EvidenceVerifier | None,
        packet: TelemetryPacket,
        device_record: DeviceRecord | None,
    ) -> bool | None:
        """Return None when evidence is absent or cannot be verified."""
        if verifier is None:
            return None
        try:
            return verifier(packet, device_record)
        except Exception:
            return None

    def assess(
        self,
        packet: TelemetryPacket,
        signature: str | None,
        *,
        now: datetime | None = None,
        behavioural_score: float | None = None,
        recovery_request_id: str | None = None,
    ) -> C4TrustAssessment:
        assessed_at = now or datetime.now(UTC)

        if assessed_at.tzinfo is None or assessed_at.utcoffset() is None:
            raise ValueError("now must be timezone-aware")

        if behavioural_score is not None:
            try:
                behavioural_score = float(behavioural_score)
            except (TypeError, ValueError) as exc:
                raise ValueError("behavioural_score must be numeric") from exc
            if not math.isfinite(behavioural_score) or not 0.0 <= behavioural_score <= 1.0:
                raise ValueError("behavioural_score must be finite and between 0 and 1")

        failures: list[str] = []
        security_unknowns: list[str] = []
        behavioural_unknowns: list[str] = []
        verified: list[str] = []
        device_record: DeviceRecord | None = None

        # Resolve identity and credential from the registry or legacy mapping.
        if self._registry is not None:
            device_record = self._registry.get(packet.device_id)
            if device_record is None:
                failures.append("DEVICE_UNKNOWN")
            elif not self._registry.is_active(packet.device_id):
                failures.append("DEVICE_NOT_ACTIVE")
            credential = device_record.credential if device_record else None
            if device_record is not None:
                if packet.zone_id != device_record.zone_id:
                    failures.append("ZONE_MISMATCH")
                else:
                    verified.append("ZONE_VALID")
                if packet.transport.gateway_id is None:
                    security_unknowns.append("HUB_ID_MISSING")
                elif packet.transport.gateway_id != device_record.hub_id:
                    failures.append("HUB_MISMATCH")
                else:
                    verified.append("HUB_VALID")

                if self._require_firmware_approval and not device_record.approved_firmware:
                    security_unknowns.append("FIRMWARE_APPROVAL_MISSING")
                elif device_record.approved_firmware:
                    if not device_record.firmware_hash or not packet.security.firmware_hash:
                        security_unknowns.append("FIRMWARE_EVIDENCE_MISSING")
                    elif device_record.firmware_hash != packet.security.firmware_hash:
                        failures.append("FIRMWARE_UNAPPROVED")
                    else:
                        verified.append("FIRMWARE_VALID")

                if self._require_config_approval and not device_record.approved_config:
                    security_unknowns.append("CONFIG_APPROVAL_MISSING")
                elif device_record.approved_config:
                    if not device_record.config_hash or not packet.security.config_hash:
                        security_unknowns.append("CONFIG_EVIDENCE_MISSING")
                    elif device_record.config_hash != packet.security.config_hash:
                        failures.append("CONFIG_MISMATCH")
                    else:
                        verified.append("CONFIG_VALID")
        else:
            credential = self._credentials.get(packet.device_id)
            if credential is None:
                failures.append("DEVICE_UNKNOWN")
            else:
                verified.append("IDENTITY_VALID")

        if device_record is not None and self._registry is not None:
            if credential is None:
                security_unknowns.append("CREDENTIAL_NOT_CONFIGURED")
            else:
                verified.append("IDENTITY_VALID")

        if credential is not None:
            if not packet.security.credential_id:
                security_unknowns.append("CREDENTIAL_ID_MISSING")
            elif not hmac.compare_digest(
                packet.security.credential_id,
                credential.credential_id,
            ):
                failures.append("CREDENTIAL_ID_MISMATCH")
            else:
                verified.append("CREDENTIAL_ID_VALID")

            for name, value in (
                ("CREDENTIAL_ISSUED_AT_INVALID", credential.issued_at),
                ("CREDENTIAL_EXPIRES_AT_INVALID", credential.expires_at),
            ):
                if value is not None and (value.tzinfo is None or value.utcoffset() is None):
                    security_unknowns.append(name)
            if (
                credential.issued_at is not None
                and credential.issued_at.tzinfo is not None
                and credential.issued_at.utcoffset() is not None
            ):
                if credential.issued_at > assessed_at:
                    failures.append("CREDENTIAL_NOT_YET_VALID")
            if (
                credential.expires_at is not None
                and credential.expires_at.tzinfo is not None
                and credential.expires_at.utcoffset() is not None
            ):
                if credential.expires_at <= assessed_at:
                    failures.append("CREDENTIAL_EXPIRED")
            if not any(
                reason.startswith("CREDENTIAL_")
                for reason in security_unknowns + failures
            ):
                verified.append("CREDENTIAL_VALID")
        elif self._registry is not None and device_record is None:
            security_unknowns.append("CREDENTIAL_UNAVAILABLE")

        if not signature:
            security_unknowns.append("SIGNATURE_MISSING")
        elif credential is None:
            security_unknowns.append("SIGNATURE_UNVERIFIABLE")
        else:
            expected = self.sign_packet(packet, credential.secret)
            if not hmac.compare_digest(expected, signature.lower()):
                failures.append("INTEGRITY_INVALID")
            else:
                verified.append("INTEGRITY_VALID")

        attestation = self._verify_evidence(
            self._attestation_verifier, packet, device_record
        )
        if attestation is True:
            verified.append("ATTESTATION_VALID")
        elif attestation is False:
            failures.append("ATTESTATION_INVALID")
        else:
            security_unknowns.append("ATTESTATION_UNVERIFIED")

        provenance = self._verify_evidence(
            self._provenance_verifier, packet, device_record
        )
        if provenance is True:
            verified.append("PROVENANCE_VALID")
        elif provenance is False:
            failures.append("PROVENANCE_INVALID")
        else:
            security_unknowns.append("PROVENANCE_UNVERIFIED")

        if packet.observed_at.tzinfo is None or packet.observed_at.utcoffset() is None:
            security_unknowns.append("TIMESTAMP_UNVERIFIABLE")
        elif packet.observed_at > assessed_at + self._future_tolerance:
            failures.append("TIMESTAMP_IN_FUTURE")
        else:
            authorized_replay = self._check_authorized_replay(
                packet, recovery_request_id
            )
            if assessed_at - packet.observed_at > self._max_age and not authorized_replay:
                failures.append("FRESHNESS_EXPIRED")
            elif authorized_replay and assessed_at - packet.observed_at > self._max_age:
                verified.append("AUTHORIZED_REPLAY_FRESHNESS_EXCEPTION")
            else:
                verified.append("FRESHNESS_VALID")

        packet_key = (packet.device_id, packet.packet_id)
        sequence_key = (packet.device_id, packet.boot_id)
        is_authorized_replay = self._check_authorized_replay(packet, recovery_request_id)
        if self._require_sequence and (packet.boot_id is None or packet.sequence_no is None):
            security_unknowns.append("SEQUENCE_EVIDENCE_MISSING")

        if not failures and not security_unknowns:
            with self._lock:
                if packet_key in self._seen_packets and not is_authorized_replay:
                    failures.append("REPLAY_DETECTED")
                if packet.sequence_no is not None:
                    previous = self._last_sequence.get(sequence_key)
                    if (
                        previous is not None
                        and packet.sequence_no <= previous
                        and not is_authorized_replay
                    ):
                        failures.append("SEQUENCE_REPLAY_DETECTED")
                if not failures and not is_authorized_replay:
                    self._seen_packets.add(packet_key)
                    if packet.sequence_no is not None:
                        self._last_sequence[sequence_key] = packet.sequence_no
                if not failures:
                    verified.append("REPLAY_CHECK_VALID")

        score = behavioural_score
        mandatory_security_pass = not failures and not security_unknowns
        if failures:
            trust_label = C4TrustLabel.UNTRUSTED
            routing_outcome = C4RoutingOutcome.REJECT
            trust_score = 0.0
        elif security_unknowns:
            trust_label = C4TrustLabel.UNKNOWN
            routing_outcome = C4RoutingOutcome.SECURITY_REVIEW
            trust_score = None
        else:
            if score is None and self._behavioural_scorer is not None:
                try:
                    score_value = self._behavioural_scorer(packet)
                    score = None if score_value is None else float(score_value)
                except Exception:
                    behavioural_unknowns.append("BEHAVIOURAL_INFERENCE_FAILED")
            if score is None and not self._allow_deterministic_only:
                behavioural_unknowns.append("BEHAVIOURAL_SCORE_UNAVAILABLE")
            if score is not None and (
                not math.isfinite(score) or not 0.0 <= score <= 1.0
            ):
                behavioural_unknowns.append("BEHAVIOURAL_SCORE_INVALID")
                score = None

            if behavioural_unknowns:
                trust_label = C4TrustLabel.UNKNOWN
                routing_outcome = C4RoutingOutcome.SECURITY_REVIEW
                trust_score = None
            elif score is not None and score >= self._suspicious_score_threshold:
                trust_label = C4TrustLabel.SUSPICIOUS
                routing_outcome = C4RoutingOutcome.QUARANTINE
                trust_score = 1.0 - score
                verified.append("BEHAVIOURAL_ANOMALY")
            else:
                trust_label = C4TrustLabel.TRUSTED
                routing_outcome = C4RoutingOutcome.ACCEPT
                trust_score = 1.0 if score is None else 1.0 - score
                if score is None:
                    verified.append("DETERMINISTIC_ONLY_POLICY")
                else:
                    verified.append("BEHAVIOURAL_NORMAL")

        return C4TrustAssessment(
            schema_version="1.0",
            assessment_id=f"C4A_{uuid4().hex}",
            packet_id=packet.packet_id,
            device_id=packet.device_id,
            trust_label=trust_label,
            trust_score=trust_score,
            mandatory_security_pass=mandatory_security_pass,
            reason_codes=list(
                dict.fromkeys(
                    verified + failures + security_unknowns + behavioural_unknowns
                )
            ),
            behavioural_score=score,
            routing_outcome=routing_outcome,
            model_version="C4_RULE_BASELINE_001",
            security_policy_version="C4_POLICY_1.1",
            generated_at=assessed_at,
            correlation_id=packet.correlation_id,
            test_run_id=packet.test_run_id,
        )

@dataclass(frozen=True)
class C4GateResult:
    """Decision and downstream eligibility produced by the C4 boundary."""

    assessment: C4TrustAssessment
    eligible_packet: TelemetryPacket | None
    quarantine_record: QuarantineRecord | None

    @property
    def eligible_for_c1(self) -> bool:
        return self.eligible_packet is not None


class C4TelemetryGate:
    """Apply C4 trust policy before a packet can be handed to C1."""

    def __init__(
        self,
        trust_service: C4TrustService,
        quarantine_manager: QuarantineManager,
    ) -> None:
        self._trust_service = trust_service
        self._quarantine_manager = quarantine_manager

    def process(
        self,
        packet: TelemetryPacket,
        signature: str | None,
        *,
        now: datetime | None = None,
        recovery_request_id: str | None = None,
    ) -> C4GateResult:
        assessment = self._trust_service.assess(
            packet,
            signature,
            now=now,
            recovery_request_id=recovery_request_id,
        )

        if (
            assessment.trust_label is C4TrustLabel.TRUSTED
            and assessment.mandatory_security_pass
            and assessment.routing_outcome is C4RoutingOutcome.ACCEPT
        ):
            return C4GateResult(
                assessment=assessment,
                eligible_packet=packet,
                quarantine_record=None,
            )

        if assessment.trust_label is C4TrustLabel.UNKNOWN:
            quarantine_reason = QuarantineReason.EVIDENCE_UNAVAILABLE
        elif assessment.trust_label is C4TrustLabel.UNTRUSTED:
            quarantine_reason = QuarantineReason.SECURITY_FAILURE
        else:
            quarantine_reason = QuarantineReason.BEHAVIOURAL_ANOMALY

        record = self._quarantine_manager.quarantine(
            packet_id=packet.packet_id,
            device_id=packet.device_id,
            trust_label=assessment.trust_label,
            routing_outcome=assessment.routing_outcome or C4RoutingOutcome.SECURITY_REVIEW,
            reason_codes=assessment.reason_codes,
            quarantine_reason=quarantine_reason,
            packet_data=packet.model_dump(mode="json"),
            assessment=assessment,
            metadata={
                "mandatory_security_pass": assessment.mandatory_security_pass,
                "security_policy_version": assessment.security_policy_version,
            },
        )
        return C4GateResult(
            assessment=assessment,
            eligible_packet=None,
            quarantine_record=record,
        )

def test_split_generation_uses_device_holdout_and_updates_split_metadata(
    tmp_path,
):

    rows = [
        {
            "device_id": f"device-{device_index}",
            "scenario": f"scenario-{packet_index % 2}",
            "trust_label": labels[device_index % len(labels)],
            "split": "legacy-row-split",

        for packet_index in range(2)
    ]
    input_path = tmp_path / "behavioral.csv"

def test_scenario_and_device_split_keeps_both_groups_disjoint_without_mutation():
    source = pd.DataFrame(
        {
            "scenario": ["A", "A", "B", "B", "C", "C"],
            "device_id": ["A1", "A2", "B1", "B2", "C1", "C2"],
        }
    )
    original = source.copy(deep=True)

    splits = ScenarioAwareSplitter(
        random_seed=3
    ).split_by_scenario_and_device(source)

    for first, second in (
        ("train", "val"),
        ("train", "test"),
        ("val", "test"),
    ):
        assert set(splits[first]["scenario"]).isdisjoint(
            splits[second]["scenario"]
        )
        assert set(splits[first]["device_id"]).isdisjoint(
            splits[second]["device_id"]
        )
    pd.testing.assert_frame_equal(source, original)

def test_quarantine_assessment_serializes_and_reloads(tmp_path):

        schema_version="1.0",

def test_untrusted_quarantine_requires_fresh_trusted_revalidation(tmp_path):
    manager = quarantine_module.QuarantineManager(
        tmp_path,
        release_authorizer=lambda reviewer_id: reviewer_id == "security-operator",
    )
    original_assessment = C4TrustAssessment(
        schema_version="1.0",
        assessment_id="C4A_ORIGINAL_001",
        packet_id="packet-review",
        device_id="device-review",
        trust_label=C4TrustLabel.UNTRUSTED,
        trust_score=0.0,
        mandatory_security_pass=False,
        reason_codes=["INTEGRITY_INVALID"],
        routing_outcome=C4RoutingOutcome.REJECT,
        generated_at=datetime.now(UTC),
    )
    record = manager.quarantine(
        packet_id=original_assessment.packet_id,
        device_id=original_assessment.device_id,
        trust_label=original_assessment.trust_label,
        routing_outcome=C4RoutingOutcome.REJECT,
        reason_codes=original_assessment.reason_codes,
        quarantine_reason=quarantine_module.QuarantineReason.SECURITY_FAILURE,
        assessment=original_assessment,
    )

    with pytest.raises(ValueError, match="fresh TRUSTED assessment"):
        manager.release(
            record.quarantine_id,
            "Manual review only",
            reviewer_id="security-operator",
        )

    revalidation = C4TrustAssessment(
        schema_version="1.0",
        assessment_id="C4A_REVALIDATED_001",
        packet_id=record.packet_id,
        device_id=record.device_id,
        trust_label=C4TrustLabel.TRUSTED,
        trust_score=1.0,
        mandatory_security_pass=True,
        reason_codes=["REVALIDATED"],
        routing_outcome=C4RoutingOutcome.ACCEPT,
        generated_at=record.quarantined_at + timedelta(seconds=1),
    )
    released = manager.release(
        record.quarantine_id,
        "Fresh security verification passed",
        review_assessment=revalidation,
        reviewer_id="security-operator",
    )

    assert released.trust_label is C4TrustLabel.TRUSTED
    assert released.assessment == original_assessment
    assert released.review_assessment == revalidation
    assert released.reviewed_by == "security-operator"

def test_c4_gate_exposes_only_trusted_packet_to_c1(tmp_path):
    packet = _runtime_packet()
    credential = DeviceCredential("cred-gate-test", b"gate-secret")
    service = C4TrustService(
        {packet.device_id: credential},
        attestation_verifier=lambda incoming, device: True,
        provenance_verifier=lambda incoming, device: True,
        behavioural_scorer=lambda incoming: 0.1,
    )
    signature = service.sign_packet(packet, credential.secret)
    gate = C4TelemetryGate(
        service,
        quarantine_module.QuarantineManager(tmp_path),
    )

    result = gate.process(packet, signature)

    assert result.eligible_for_c1
    assert result.eligible_packet is packet
    assert result.quarantine_record is None
    assert result.assessment.trust_label is C4TrustLabel.TRUSTED


def test_c4_gate_sends_unknown_and_untrusted_packets_to_quarantine(tmp_path):
    packet = _runtime_packet()
    credential = DeviceCredential("cred-gate-test", b"gate-secret")
    service = C4TrustService(
        {packet.device_id: credential},
        allow_deterministic_only=True,
    )
    quarantine = quarantine_module.QuarantineManager(tmp_path)
    gate = C4TelemetryGate(service, quarantine)

    unknown_result = gate.process(
        packet,
        service.sign_packet(packet, credential.secret),
    )
    untrusted_result = gate.process(packet, "invalid-signature")

    assert unknown_result.assessment.trust_label is C4TrustLabel.UNKNOWN
    assert (
        unknown_result.assessment.routing_outcome
        is C4RoutingOutcome.SECURITY_REVIEW
    )
    assert unknown_result.eligible_packet is None
    assert unknown_result.quarantine_record is not None
    assert untrusted_result.assessment.trust_label is C4TrustLabel.UNTRUSTED
    assert untrusted_result.assessment.routing_outcome is C4RoutingOutcome.REJECT
    assert untrusted_result.eligible_packet is None
    assert quarantine.count() == 2
